+int rpc_hs_server_accept(rpc_hs_session_t *session, int fd,
+			 const rpc_hs_server_config_t *config,
+			 rpc_hs_result_t *result)
+{
+	rpc_hs_message_t first = { 0 };
+	uint8_t payload[RPC_HS_MAX_PAYLOAD];
+	int ret;
+
+	if (!session || fd < 0 || !config || !result)
+		return -EINVAL;
+	rpc_hs_session_init_plain(session, fd);
+	ret = rpc_hs_recv(session, &first, payload, sizeof(payload));
+	if (ret != 0) {
+		rpc_hs_send_error(session, RPC_HS_OP_NEGOTIATE,
+				  ret > 0 ? (uint16_t)ret : RPC_HS_ERR_FRAME);
+		return ret;
+	}
+
+	if (first.operation == RPC_HS_OP_TIME) {
+		ret = rpc_hs_send_time_response(session, (uint64_t)time(NULL));
+		memset(result, 0, sizeof(*result));
+		result->result = RPC_HS_OK_TIME;
+		return ret;
+	}
+
+	if (first.operation != RPC_HS_OP_NEGOTIATE) {
+		rpc_hs_send_error(session, first.operation,
+				  RPC_HS_ERR_BAD_OPERATION);
+		return RPC_HS_ERR_BAD_OPERATION;
+	}
+	if ((first.flags & RPC_HS_F_MTLS_REQUEST) &&
+	    !tls_cert_server_available()) {
+		rpc_hs_send_error(session, RPC_HS_OP_NEGOTIATE,
+				  RPC_HS_ERR_MTLS_UNAVAILABLE);
+		return RPC_HS_ERR_MTLS_UNAVAILABLE;
+	}
+
+	ret = rpc_hs_server_respond(session, &first, config->mtls_required,
+				    config->algorithm, config->ca_cn, result);
+	if (ret != 0)
+		return ret;
+	if (result->result == RPC_HS_OK_MTLS) {
+		SSL *ssl = tls_cert_server_handshake(fd, NULL);
+		if (!ssl)
+			return RPC_HS_ERR_MTLS_UNAVAILABLE;
+		rpc_hs_session_init_tls(session, fd, ssl);
+	}
+	return 0;
+}
